  async function expConvert(file){
    const ext=(file.name.split('.').pop()||'').toLowerCase();
    if(['txt','md','log','json','xml','html','htm'].includes(ext))return {kind:ext.toUpperCase(),text:`# ${file.name}\n\n${await file.text()}`};
    if(ext==='pdf')return {kind:'PDF',text:await expPdf(file)};
    if(ext==='docx')return {kind:'WORD',text:await expDocx(file)};
    if(['xlsx','xls','csv'].includes(ext))return {kind:'EXCEL',text:await expSheet(file)};
    if(ext==='pptx')return {kind:'POWERPOINT',text:await expPptx(file)};
    if(file.type.startsWith('image/')||file.type.startsWith('video/'))return {kind:'MEDIA',blob:file};
    return {kind:'FILE',text:`# ${file.name}\n\nUnsupported rich conversion for this type.\n\nMIME: ${file.type||'unknown'}\nSize: ${Math.round(file.size/1024)} KB`};
  }
